\begin{frame}{1. Executive Overview \& Problem Formulation}
  \begin{block}{The Federated Causal Discovery Challenge}
    Discovering a global Causal DAG $G^* = (\mathbf{V}, \mathbf{E})$ across $K$ sovereign, non-trusting federated agents under strict privacy and finite interventional budget constraints.
  \end{block}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Key Constraints \& Challenges:}
      \begin{itemize}
        \item \textbf{No Data Pooling:} Privacy laws (GDPR/HIPAA) prohibit sharing raw samples $\mathbf{S}$.
        \item \textbf{Markov Equivalence (MEC):} Passive observation cannot orient edges; active hard interventions $do(X_i = c)$ are required.
        \item \textbf{Resource Limits:} Interventions incur physical/budgetary costs $B_{k,t}$.
      \end{itemize}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \textbf{Variable Taxonomy ($\mathbf{V} = [Z_1, X_1, X_2, Z_2]$):}
      \begin{itemize}
        \item \textbf{Private Local ($Z_k$):} Visible and controllable exclusively by Agent $k$.
        \item \textbf{Boundary Interfaces ($X_k$):} Shared interface across jurisdictions.
        \item \textbf{Dec-POMDP Paradigm:} Solved via Disjoint Independent PPO (IPPO).
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE 3: Key Architectural Innovations
%-------------------------------------------------------
\begin{frame}{2. Key Architectural Innovations}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Skew-Symmetric Tournament Head}
        Decomposes edge logits into symmetric skeleton and anti-symmetric tournament streams:
        $$\text{Logit}_{i \to j} = S_\theta(e_i, e_j) + \mathcal{O}_\phi(e_i, e_j) + \gamma \mathbf{A}_{ij}$$
        \begin{itemize}
          \item \textbf{Algebraic Guarantee:} Eliminates 2-cycle conflicts ($X_i \rightleftarrows X_j$) by construction ($\text{Logit}_{i \to j} - \text{Logit}_{j \to i} \equiv 2\mathcal{O}_\phi + 2\gamma \mathbf{A}_{ij}$).
          \item Prevents static prior memorization by coupling to variance asymmetry $\mathbf{A}_{ij}$.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{Disjoint IPPO \& Full GPU JIT Pipeline}
        \begin{itemize}
          \item \textbf{Disjoint Parameters:} Sovereign $(\theta_k, \phi_k)$ without weight sharing to preserve federated privacy.
          \item \textbf{Hierarchical Action Space:} Macro (Local, Peer Request, NOOP) + Micro target.
          \item \textbf{$>300\times$ GPU Speedup:} Zero-sync Gumbel action sampling, matrix-power cycle detection $\text{Tr}(A^k)$, and fused SCM kernels ($>316$ ep/s).
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{4. Empirical Benchmark Results \& Key Insights}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Standardized 8-Experiment Benchmark}
        Standardized across 3 fixed random seeds ($42, 43, 44$):
        \begin{itemize}
          \item \textbf{EXP-1 (Sanity):} 100\% convergence on single topology overfits ($\text{SHD} \to 0$).
          \item \textbf{EXP-2/5 (Curriculum):} 3-stage curriculum reduces multi-topology SHD variance by $>45\%$.
          \item \textbf{EXP-3 (Head Ablation):} Skew-symmetric head eliminates 2-cycle penalties completely vs. unconstrained MLPs.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{Exploration \& Budget Stress Tests}
        \begin{itemize}
          \item \textbf{EXP-4 (Curiosity Sweep):} $\beta = 0.05$ achieves optimal balance between interventional discovery and budget conservation.
          \item \textbf{EXP-6 (Budget Scarcity):} Agents prioritize high-asymmetry boundary interventions ($X_1 \to X_2$) when budgets are constrained ($B \le 2$).
          \item \textbf{Disjoint IPPO vs. Baselines:} Outperforms Random and Round-Robin statistical baselines in both SHD and F1 score.
        \end{itemize}
      \end{block}
    \end{column}

% SLIDE 6: Current Bottlenecks & Limitations
%-------------------------------------------------------
\begin{frame}{5. Current Bottlenecks \& Limitations}
  \begin{alertblock}{Open Challenges \& Failure Modes}
    \begin{enumerate}
      \item \textbf{Non-Linear ANM Mechanisms (EXP-7):} Current interventional variance shift asymmetry $\mathbf{A}_{ij}$ assumes linear Gaussian mechanisms. Non-linear transformations cause variance ratio misalignments.
      \item \textbf{Observational Noise Sensitivity (EXP-8):} Low sample counts ($N < 100$) or high noise levels $\sigma_i^2 > 1.0$ distort covariance estimates $\Sigma_{\text{obs}}$, leading to incorrect initial skeleton priors.
      \item \textbf{Scalability Constraints ($d=4, K=2 \to d \gg 4, K > 2$):} Higher-dimensional graphs exhibit combinatorial explosion in cross-jurisdictional boundary edge combinations and cycle check operations.
      \item \textbf{Heterogeneous Agent Budgets:} Asymmetric budget allocations ($B_1 \neq B_2$) occasionally result in uncooperative peer intervention request deadlocks.
    \end{enumerate}
  \end{alertblock}
\end{frame}
